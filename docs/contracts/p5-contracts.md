## state.ts
```ts
export type WatchdogState = "idle" | "armed" | "inspecting" | "cleared" | "steering" | "resuming" | "halted-stalemate" | "failed";
export type AgentState = "queued" | "dispatched-working" | "waiting-architect" | "in-review" | `edit-round-${number}` | "stalemate" | "harvesting" | "done-verified" | "done-unverified" | "authoring-workflow" | "repairing-workflow" | "compacting" | "inspecting-compaction" | "resuming" | "held-spend" | "blocked-guard" | "system-run" | "watchdog-failed" | "uncertain-launch" | "failed" | "cancelled" | "failed-review";
export interface WatchdogFinding { id: string; category: string; summary: string; paths: string[]; severity: "blocker" | "major" | "minor" | "info"; source: "inspector" | "child-card" | "rule"; agentId?: string; ts: string }
export interface WatchdogMachine { state: WatchdogState; findings: WatchdogFinding[]; identityRuns: Record<string, number>; inspections: number; spendUsd: number; lastTransition: string; stalemateRepeats: number }
export function createMachine(settings: { stalemateRepeats: number }): WatchdogMachine
export function transition(m: WatchdogMachine, to: WatchdogState, note?: string): WatchdogMachine   // validates the edges of §5.5 (idle→armed, armed→inspecting, inspecting→cleared|steering|resuming|halted-stalemate|failed, any→idle on disarm; invalid edge throws)
export function noteFinding(m: WatchdogMachine, f: Omit<WatchdogFinding, "id" | "ts">): { machine: WatchdogMachine; stalemate: boolean }   // identity = sha256(category + summary + sorted paths) (use elevation.findingIdentity when present); consecutive identical identity count ≥ stalemateRepeats → stalemate true and state halted-stalemate
export function mapChildCard(card: unknown): WatchdogFinding | undefined   // pi-subagents `subagent_watchdog_warning` details {severity, category, summary, evidence, identity, state} → finding (severity blocker|concern→major|note→minor)
```

## compaction.ts
```ts
export interface CompactionEvent { reason: "manual" | "threshold" | "overflow"; willRetry: boolean; signal?: AbortSignal; preparation: { firstKeptEntryId?: string; tokensBefore?: number }; branchEntries?: unknown[]; customInstructions?: string }
export interface InspectorDeps { inspect(prompt: string, opts: { timeoutMs: number; signal?: AbortSignal; model: string; thinking: string }): Promise<{ ok: boolean; text: string; usage?: { tokensIn: number; tokensOut: number; costUsd: number; tpsSeconds: number }; error?: string }> }
export interface CompactionDeps extends InspectorDeps { store: RunStore; runDir?: string; settings: WatchdogSettings; ledger(row: { origin: "compaction-inspector"; model: string; costUsd: number; tokensIn: number; tokensOut: number; ok: boolean }): void; entriesText(entries: unknown[] | undefined): string /* host renders the entries about to be summarized as text, bounded to 24,000 chars */; now?(): number }
export interface CompactionOutcome { mode: "custom" | "default" | "state-only"; summary?: string; stateBlock?: StateBlock; badge?: "watchdog-failed"; inspectorMs?: number; note: string }
export async function handleBeforeCompact(event: CompactionEvent, deps: CompactionDeps): Promise<CompactionOutcome>
```
Rules: settings.onCompaction "off" or no runDir (no active run/plan) → mode "default" (the host returns undefined so Pi summarizes as usual); reason "overflow" or willRetry → mode "state-only" (summary = state block only, no inspector — the host returns a custom compaction with that summary); "summary-only" → state-only as well; "halt-inspect" on manual|threshold → build the state block synchronously, then run the inspector with USER_PROMPT_WATCHDOG_COMPACTION.md (state block + the bounded entries text), bounded by settings.inspectorTimeoutMs (default 20000; a `timeoutMs` override in event/customInstructions "titan:inspector=45000" may raise it for manual compactions), honouring event.signal; inspector ok → mode "custom" with summary = stateBlock.text + "\n\n## narrative\n" + inspector text; inspector timeout/failure → mode "state-only" with badge "watchdog-failed" and note; never cancel. Every inspector call → deps.ledger row (origin compaction-inspector) whether ok or not. Record events on the store when runDir is set: `compaction.before {reason, mode, badge?, stateSha256}`; `handleAfterCompact(event: { reason; fromExtension; summaryHash? }, deps)` records `compaction.done {summaryHash, fromExtension}` and returns the agent-state patch (compacting → dispatched-working); `handleCompactFailed(event, deps)` records `compaction.failed`.

## preempt.ts (child pre-emption + logical clearing)
```ts
export interface ChildUsage { agentId: string; ctxTokens: number; contextWindow: number; compactionSeen?: boolean }
export function shouldPreempt(u: ChildUsage, fraction: number): boolean   // ctxTokens / contextWindow ≥ fraction, or compactionSeen
export interface PreemptDeps extends InspectorDeps { store: RunStore; runDir: string; settings: WatchdogSettings; ledger(row: { origin: "watchdog"; model: string; costUsd: number; tokensIn: number; tokensOut: number; ok: boolean }): void; transcriptTail(agentId: string, maxChars: number): string; diff?(): string; architectModel: string; architectThinking: string }
export type PreemptDecision = { action: "logical-clear"; checkpointSessionId: string; note: string } | { action: "resume-fresh"; model: string; thinking: string; resumePrompt: string; note: string } | { action: "failed"; note: string };
export async function inspectPreempted(agentId: string, block: StateBlock, deps: PreemptDeps): Promise<PreemptDecision>
```
Inspector prompt = SYSTEM_PROMPT_WATCHDOG.md + the state block + the transcript tail (≤ 24,000 chars) and asks for a strict verdict `{"verdict": "clean" | "loss" | "hallucination", "reasons": [...], "carry": "<what the next session must know>"}` (parse leniently: last JSON object; missing → "loss"). clean → logical-clear: agents/<id>.json gets `logicalCleared: {at, priorSessionRef, reason}` (via store.upsertAgent patch — add the field on AgentRecord as optional; run-store.ts is NOT yours: put the patch in an `extra` record field the store already accepts (upsertAgent takes a Partial<AgentRecord> — if it rejects unknown fields, store the flag in an event `watchdog.logical_clear` only and say so), a fresh checkpoint session id (randomUUID) returned for the host to spawn with `sessionId`; loss|hallucination → resume-fresh with USER_PROMPT_RESUME.md rendered (state block + diff + findings + carry) on deps.architectModel; inspector failure → "failed" (host marks watchdog-failed and leaves the child halted). Never a transcript replay into another model. Ledger every inspection.
